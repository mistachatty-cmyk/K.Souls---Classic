const handleTouchStart = (e: TouchEvent) => {
  e.preventDefault();
  
  // If there are no fingers specifically on the canvas, ignore it.
  if (e.targetTouches.length === 0) return;

  const rect = canvas.getBoundingClientRect();
  
  // ONLY look at the first finger that is on the canvas itself
  const touch = e.targetTouches[0];
  
  const tx = (touch.clientX - rect.left) * (canvas.width / rect.width);
  const ty = (touch.clientY - rect.top) * (canvas.height / rect.height);
  
  mouseRef.current.x = tx;
  mouseRef.current.y = ty;
  mouseRef.current.isDown = true;

  // ... (The rest of your godAction and spawning logic goes here) ...
};

const handleTouchMove = (e: TouchEvent) => {
  e.preventDefault();
  
  // Game screen movement tracking
  if (screenRef.current === "game" && stateRef.current.running && !stateRef.current.paused) {
    if (e.targetTouches.length === 0) return;
    
    const rect = canvas.getBoundingClientRect();
    const touch = e.targetTouches[0]; // Again, only track canvas touches
    
    mouseRef.current.x = (touch.clientX - rect.left) * (canvas.width / rect.width);
    mouseRef.current.y = (touch.clientY - rect.top) * (canvas.height / rect.height);
  }
};

if (godAction === "area_lock" && state.dragStartX !== undefined) {
   const sx = state.dragStartX;
   const sy = state.dragStartY;
   const minX = Math.min(sx, mx), maxX = Math.max(sx, mx);
   const minY = Math.min(sy, my), maxY = Math.max(sy, my);
   
   let capturedCount = 0;

   // 1. Lock the characters
   for (const c of dataRef.chars.current) {
      if (!c.dead && c.x >= minX && c.x <= maxX && c.y >= minY && c.y <= maxY) {
         c.isLocked = true; 
         c.lockedX = c.x; 
         c.lockedY = c.y; 
         c.lockTimer = 3000;
         capturedCount++;
      }
   }

   // 2. OPTIONAL VISUAL EFFECT: Area Lock Confirmation
   // We respect the Sandbox Philosophy by checking if juice is enabled
   if (state.juice && state.juice !== "off") {
      const width = maxX - minX;
      const height = maxY - minY;
      
      // Only spawn effects if a valid box was drawn
      if (width > 10 && height > 10) {
         // Use the blue color matching the Area Lock preview
         const effectColor = "#3b82f6"; 
         
         // Spawn an explosion at the 4 corners of the locked area
         spawnExplosion(minX, minY, effectColor);
         spawnExplosion(maxX, minY, effectColor);
         spawnExplosion(minX, maxY, effectColor);
         spawnExplosion(maxX, maxY, effectColor);

         // If we are on "SUPER" juice and caught someone, add a center burst!
         if (state.juice === "super" && capturedCount > 0) {
            const centerX = minX + (width / 2);
            const centerY = minY + (height / 2);
            spawnExplosion(centerX, centerY, "#ffd700"); // Gold burst in the middle
         }
      }
   }
}
